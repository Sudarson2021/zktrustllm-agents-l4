  terminal/.style={draw, rounded corners=1.5pt, align=center, minimum height=7mm, minimum width=17mm, fill=gray!10},

\node[state] (validate) {Validate\\schema + policy};
\node[state, right=of validate] (submit) {Submit\\HUMAN request};
\node[guard, right=of submit] (decision) {Operator\\decision};
\node[state, right=of decision] (ledger) {Simulated\\ledger write};
\node[terminal, below=of decision] (stop) {Reject / expire\\no actuation};

\draw[flow] (validate) -- (submit);
\draw[flow] (validate) |- node[pos=.25,left]{invalid / NEVER} (stop);
\draw[flow] (submit) -- (decision);
\draw[flow] (decision) -- node[above,font=\scriptsize]{approve} (ledger);
\draw[flow] (decision) -- node[right]{reject / timeout} (stop);

\caption{HUMAN-gate state transition. Only a valid HUMAN proposal followed by
an explicit approval reaches the simulated ledger; every invalid, rejected, or
expired path terminates without actuation. The ledger and actuation are
simulated, not a production O-RAN deployment.}
